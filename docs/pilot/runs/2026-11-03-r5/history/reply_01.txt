import json
import os


def collect(loop_dir):
    if not os.path.isdir(loop_dir):
        return ([], [])
    
    rows = []
    bad = []
    
    for entry in os.listdir(loop_dir):
        subdir = os.path.join(loop_dir, entry)
        if not os.path.isdir(subdir):
            continue
        status_file = os.path.join(subdir, "status.json")
        if not os.path.isfile(status_file):
            continue
        try:
            with open(status_file, 'r', encoding='utf-8') as f:
                data = json.load(f)
            if not isinstance(data, dict):
                bad.append(entry)
                continue
            name = data.get('name', '')
            state = data.get('state', '')
            reason = data.get('reason', '')
            coder_calls = data.get('coder_calls', 0)
            updated = data.get('updated', '')
            
            reason = reason[:80]
            
            rows.append({
                'name': name,
                'state': state,
                'reason': reason,
                'coder_calls': coder_calls,
                'updated': updated
            })
        except Exception:
            bad.append(entry)
    
    rows.sort(key=lambda r: r['name'])
    rows.sort(key=lambda r: r['updated'], reverse=True)
    bad.sort()
    
    return (rows, bad)


def main(argv):
    if not argv:
        loop_dir = "_loop"
    else:
        loop_dir = argv[0]
    
    rows, bad = collect(loop_dir)
    
    for row in rows:
        print(f"{row['name']} | {row['state']} | {row['coder_calls']} | {row['reason']}")
    
    if bad:
        print("unreadable: " + ", ".join(bad))
    
    return 0

import json
import os
import sys
import tempfile
import unittest
from io import StringIO

# Add the source directory to path
sys.path.insert(0, os.path.join(os.path.dirname(__file__), '..', 'scripts', 'source'))
import loop_status


class TestCollect(unittest.TestCase):
    def test_nonexistent_dir(self):
        rows, bad = loop_status.collect("/nonexistent/path")
        self.assertEqual(rows, [])
        self.assertEqual(bad, [])
    
    def test_empty_dir(self):
        with tempfile.TemporaryDirectory() as tmpdir:
            rows, bad = loop_status.collect(tmpdir)
            self.assertEqual(rows, [])
            self.assertEqual(bad, [])
    
    def test_valid_status(self):
        with tempfile.TemporaryDirectory() as tmpdir:
            subdir = os.path.join(tmpdir, "job1")
            os.makedirs(subdir)
            status = {
                "name": "job1",
                "state": "done",
                "reason": "All good",
                "coder_calls": 5,
                "updated": "2024-01-01T10:00:00"
            }
            with open(os.path.join(subdir, "status.json"), 'w') as f:
                json.dump(status, f)
            
            rows, bad = loop_status.collect(tmpdir)
            self.assertEqual(len(rows), 1)
            self.assertEqual(rows[0]['name'], "job1")
            self.assertEqual(rows[0]['state'], "done")
            self.assertEqual(rows[0]['reason'], "All good")
            self.assertEqual(rows[0]['coder_calls'], 5)
            self.assertEqual(rows[0]['updated'], "2024-01-01T10:00:00")
            self.assertEqual(bad, [])
    
    def test_missing_fields(self):
        with tempfile.TemporaryDirectory() as tmpdir:
            subdir = os.path.join(tmpdir, "job1")
            os.makedirs(subdir)
            status = {}
            with open(os.path.join(subdir, "status.json"), 'w') as f:
                json.dump(status, f)
            
            rows, bad = loop_status.collect(tmpdir)
            self.assertEqual(len(rows), 1)
            self.assertEqual(rows[0]['name'], "")
            self.assertEqual(rows[0]['state'], "")
            self.assertEqual(rows[0]['reason'], "")
            self.assertEqual(rows[0]['coder_calls'], 0)
            self.assertEqual(rows[0]['updated'], "")
    
    def test_reason_truncation(self):
        with tempfile.TemporaryDirectory() as tmpdir:
            subdir = os.path.join(tmpdir, "job1")
            os.makedirs(subdir)
            long_reason = "x" * 100
            status = {
                "name": "job1",
                "state": "done",
                "reason": long_reason,
                "coder_calls": 0,
                "updated": "2024-01-01"
            }
            with open(os.path.join(subdir, "status.json"), 'w') as f:
                json.dump(status, f)
            
            rows, bad = loop_status.collect(tmpdir)
            self.assertEqual(len(rows[0]['reason']), 80)
            self.assertEqual(rows[0]['reason'], "x" * 80)
    
    def test_invalid_json(self):
        with tempfile.TemporaryDirectory() as tmpdir:
            subdir = os.path.join(tmpdir, "badjob")
            os.makedirs(subdir)
            with open(os.path.join(subdir, "status.json"), 'w') as f:
                f.write("not json")
            
            rows, bad = loop_status.collect(tmpdir)
            self.assertEqual(rows, [])
            self.assertEqual(bad, ["badjob"])
    
    def test_non_object_json(self):
        with tempfile.TemporaryDirectory() as tmpdir:
            subdir = os.path.join(tmpdir, "badjob")
            os.makedirs(subdir)
            with open(os.path.join(subdir, "status.json"), 'w') as f:
                json.dump([1, 2, 3], f)
            
            rows, bad = loop_status.collect(tmpdir)
            self.assertEqual(rows, [])
            self.assertEqual(bad, ["badjob"])
    
    def test_sorting(self):
        with tempfile.TemporaryDirectory() as tmpdir:
            subdir1 = os.path.join(tmpdir, "job1")
            os.makedirs(subdir1)
            status1 = {
                "name": "job1",
                "state": "done",
                "reason": "",
                "coder_calls": 0,
                "updated": "2024-01-01"
            }
            with open(os.path.join(subdir1, "status.json"), 'w') as f:
                json.dump(status1, f)
            
            subdir2 = os.path.join(tmpdir, "job2")
            os.makedirs(subdir2)
            status2 = {
                "name": "job2",
                "state": "done",
                "reason": "",
                "coder_calls": 0,
                "updated": "2024-01-02"
            }
            with open(os.path.join(subdir2, "status.json"), 'w') as f:
                json.dump(status2, f)
            
            rows, bad = loop_status.collect(tmpdir)
            self.assertEqual(len(rows), 2)
            self.assertEqual(rows[0]['name'], "job2")
            self.assertEqual(rows[1]['name'], "job1")

        with tempfile.TemporaryDirectory() as tmpdir:

    def test_bad_sorted(self):
        with tempfile.TemporaryDirectory() as tmpdir:
            for name in ["c_bad", "a_bad", "b_bad"]:
                subdir = os.path.join(tmpdir, name)
                os.makedirs(subdir)
                with open(os.path.join(subdir, "status.json"), 'w') as f:
                    f.write("invalid")
            
            rows, bad = loop_status.collect(tmpdir)
            self.assertEqual(bad, ["a_bad", "b_bad", "c_bad"])
    
    def test_ignores_deep_files(self):
        with tempfile.TemporaryDirectory() as tmpdir:
            subdir = os.path.join(tmpdir, "job1")
            os.makedirs(os.path.join(subdir, "deep"))
            with open(os.path.join(subdir, "deep", "status.json"), 'w') as f:

            rows, bad = loop_status.collect(tmpdir)
            self.assertEqual(rows, [])
            self.assertEqual(bad, [])
    
    def test_ignores_non_dir_entries(self):
        with tempfile.TemporaryDirectory() as tmpdir:

            rows, bad = loop_status.collect(tmpdir)
            self.assertEqual(rows, [])
            self.assertEqual(bad, [])


class TestMain(unittest.TestCase):

    def test_main_with_args(self):
        import unittest.mock as mock
        with mock.patch.object(loop_status, 'collect', return_value=([], [])) as mock_collect:
            with mock.patch('sys.stdout', new=StringIO()) as fake_out:
                result = loop_status.main(["/some/path"])
                self.assertEqual(result, 0)
                mock_collect.assert_called_once_with("/some/path")
    
    def test_main_prints_rows(self):
        import unittest.mock as mock
        rows = [
            {"name": "job1", "state": "done", "reason": "ok", "coder_calls": 3, "updated": "2024-01-01"},
            {"name": "job2", "state": "running", "reason": "working", "coder_calls": 5, "updated": "2024-01-02"},

        with mock.patch.object(loop_status, 'collect', return_value=(rows, [])):

                self.assertEqual(result, 0)

                output = fake_out.getvalue()
