```python
import json
import os
import re
import time
import tempfile


_KEY_RE = re.compile(r'^[A-Za-z0-9_-]{8,128}$')


class IdempotencyStore:
    def __init__(self, folder, ttl_s=86400, clock=time.time):
        self._folder = folder
        self._ttl_s = ttl_s
        self._clock = clock
        if not os.path.isdir(folder):
            os.makedirs(folder)

    def _path_for(self, key):
        return os.path.join(self._folder, key + '.json')

    def _read(self, path):
        try:
            with open(path, 'r') as f:
                data = json.load(f)
            return data
        except (OSError, ValueError):
            return None

    def _write(self, path, data):
        fd, tmp_path = tempfile.mkstemp(dir=self._folder)
        try:
            with os.fdopen(fd, 'w') as f:
                json.dump(data, f)
            os.replace(tmp_path, path)
        except Exception:
            if os.path.exists(tmp_path):
                os.remove(tmp_path)
            raise

    def get(self, key):
        if not isinstance(key, str) or not _KEY_RE.match(key):
            raise ValueError(f"Invalid key: {key!r}")
        path = self._path_for(key)
        data = self._read(path)
        if data is None:
            return None
        ts = data.get('ts')
        if ts is None:
            return None
        if self._clock() - ts > self._ttl_s:
            return None
        return data.get('response')

    def put(self, key, response):
        if not isinstance(key, str) or not _KEY_RE.match(key):
            raise ValueError(f"Invalid key: {key!r}")
        path = self._path_for(key)
        data = self._read(path)
        if data is not None:
            ts = data.get('ts')
            if ts is not None and self._clock() - ts <= self._ttl_s:
                return data.get('response')
        new_data = {'ts': self._clock(), 'response': response}
        self._write(path, new_data)
        return response
```

```python
import os

class TestIdempotencyStore(unittest.TestCase):
    def setUp(self):
        self.tmp = tempfile.mkdtemp()
        self.clock_calls = [0.0]
        def clock():
            return self.clock_calls[0]
        self.clock = clock
        self.store = IdempotencyStore(self.tmp, ttl_s=100, clock=clock)

    def tearDown(self):
        import shutil
        shutil.rmtree(self.tmp, ignore_errors=True)

    def test_put_get_roundtrip(self):
        resp = {'status': 200, 'body': 'hello'}
        self.assertEqual(self.store.put('valid_key_1', resp), resp)
        self.assertEqual(self.store.get('valid_key_1'), resp)

    def test_first_write_wins(self):
        resp1 = {'v': 1}
        resp2 = {'v': 2}
        self.store.put('first_key_1', resp1)
        self.store.put('first_key_1', resp2)
        self.assertEqual(self.store.get('first_key_1'), resp1)

    def test_get_missing(self):
        self.assertIsNone(self.store.get('missing_key_1'))

    def test_invalid_key_short(self):
        with self.assertRaises(ValueError):
            self.store.put('short', {'a': 1})
        with self.assertRaises(ValueError):
            self.store.get('short')

    def test_invalid_key_long(self):
        long_key = 'a' * 129
        with self.assertRaises(ValueError):
            self.store.put(long_key, {'a': 1})

    def test_invalid_key_chars(self):
        with self.assertRaises(ValueError):
            self.store.put('bad key!!!', {'a': 1})
        with self.assertRaises(ValueError):
            self.store.put('key with spaces', {'a': 1})

    def test_ttl_expiry(self):
        resp = {'x': 1}
        self.store.put('exp_key_1', resp)
        self.clock_calls[0] = 200  # beyond ttl of 100
        self.assertIsNone(self.store.get('exp_key_1'))
        # put after expiry overwrites
        new_resp = {'x': 2}
        result = self.store.put('exp_key_1', new_resp)
        self.assertEqual(result, new_resp)
        self.clock_calls[0] = 200
        self.assertEqual(self.store.get('exp_key_1'), new_resp)

    def test_unreadable_json_treated_as_absent(self):
        path = os.path.join(self.tmp, 'corrupt_key.json')
        with open(path, 'w') as f:
            f.write('not json')
        self.assertIsNone(self.store.get('corrupt_key'))
        # put should overwrite
        resp = {'ok': True}
        result = self.store.put('corrupt_key', resp)
        self.assertEqual(result, resp)
        self.assertEqual(self.store.get('corrupt_key'), resp)

    def test_folder_created(self):
        new_dir = os.path.join(self.tmp, 'sub', 'folder')
        store = IdempotencyStore(new_dir, clock=self.clock)
        store.put('nested_key_1', {'a': 1})
        self.assertEqual(store.get('nested_key_1'), {'a': 1})

    def test_default_ttl(self):
        # default ttl 86400

        self.clock_calls[0] = 100000
        store.put('default_key', {'b': 2})
        self.clock_calls[0] = 100000 + 86399
        self.assertEqual(store.get('default_key'), {'b': 2})
        self.clock_calls[0] = 100000 + 86401
        self.assertIsNone(store.get('default_key'))

    def test_file_written_whole(self):
        # verify os.replace semantics by reading file directly

        path = os.path.join(self.tmp, 'atomic_key_1.json')

        self.assertEqual(data['response'], {'c': 3})
        self.assertIn('ts', data)
```
