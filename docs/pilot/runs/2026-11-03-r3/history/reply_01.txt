import json
import os
import sys

def collect(loop_dir):
    if not os.path.isdir(loop_dir):
        return [], []
    
    rows = []
    bad = []
    
    for entry in os.listdir(loop_dir):
        sub_dir = os.path.join(loop_dir, entry)
        if not os.path.isdir(sub_dir):
            continue
        status_file = os.path.join(sub_dir, "status.json")
        if not os.path.isfile(status_file):
            continue
        try:
            with open(status_file, "r", encoding="utf-8") as f:
                data = json.load(f)
            if not isinstance(data, dict):
                raise ValueError("Not a JSON object")
        except Exception:
            bad.append(entry)
            continue
        
        name = data.get("name", "")
        if name is None:
            name = ""
        else:
            name = str(name)
            
        state = data.get("state", "")
        if state is None:
            state = ""
        else:
            state = str(state)
            
        reason = data.get("reason", "")
        if reason is None:
            reason = ""
        else:
            reason = str(reason)
            
        updated = data.get("updated", "")
        if updated is None:
            updated = ""
        else:
            updated = str(updated)
            
        coder_calls = data.get("coder_calls", 0)
        if coder_calls is None:
            coder_calls = 0
        else:
            try:
                coder_calls = int(coder_calls)
            except (ValueError, TypeError):
                coder_calls = 0
        
        # Truncate reason to first 80 characters
        reason = reason[:80]
        
        rows.append({
            "name": name,
            "state": state,
            "reason": reason,
            "coder_calls": coder_calls,
            "updated": updated
        })
    
    # Sort by name first, then stable sort by updated descending
    rows.sort(key=lambda r: r["name"])
    rows.sort(key=lambda r: r["updated"], reverse=True)
    
    bad.sort()
    return rows, bad

def main(argv):
    loop_dir = "_loop"
    if len(argv) > 1:
        loop_dir = argv[1]
    rows, bad = collect(loop_dir)
    for row in rows:
        print(f"{row['name']} | {row['state']} | {row['coder_calls']} | {row['reason']}")
    if bad:
        print("unreadable: " + ", ".join(bad))
    return 0

if __name__ == "__main__":
    sys.exit(main(sys.argv))

import json
import os
import sys
import tempfile
import unittest
from io import StringIO

# Adjust path to import the module
test_dir = os.path.dirname(__file__)
module_dir = os.path.join(test_dir, '..', 'scripts', 'source')
sys.path.insert(0, module_dir)
import loop_status

class TestLoopStatus(unittest.TestCase):
    
    def test_collect_nonexistent_dir(self):
        rows, bad = loop_status.collect("/nonexistent/path/for/sure")
        self.assertEqual(rows, [])
        self.assertEqual(bad, [])
    
    def test_collect_empty_dir(self):
        with tempfile.TemporaryDirectory() as tmpdir:
            rows, bad = loop_status.collect(tmpdir)
            self.assertEqual(rows, [])
            self.assertEqual(bad, [])
    
    def test_collect_no_subdirs(self):
        with tempfile.TemporaryDirectory() as tmpdir:
            with open(os.path.join(tmpdir, "file.txt"), "w") as f:
                f.write("test")
            rows, bad = loop_status.collect(tmpdir)
            self.assertEqual(rows, [])
            self.assertEqual(bad, [])
    
    def test_collect_subdir_without_status(self):
        with tempfile.TemporaryDirectory() as tmpdir:
            sub1 = os.path.join(tmpdir, "job1")
            os.makedirs(sub1)
            rows, bad = loop_status.collect(tmpdir)
            self.assertEqual(rows, [])
            self.assertEqual(bad, [])
    
    def test_collect_ignores_deep_files(self):
        with tempfile.TemporaryDirectory() as tmpdir:
            sub1 = os.path.join(tmpdir, "job1")
            os.makedirs(sub1)
            deep = os.path.join(sub1, "deep")
            os.makedirs(deep)
            status = {"name": "deep", "state": "done", "reason": "ok", "coder_calls": 0, "updated": "2023-01-01"}
            with open(os.path.join(deep, "status.json"), "w") as f:
                json.dump(status, f)
            status2 = {"name": "job1", "state": "done", "reason": "ok", "coder_calls": 0, "updated": "2023-01-01"}
            with open(os.path.join(sub1, "status.json"), "w") as f:
                json.dump(status2, f)
            rows, bad = loop_status.collect(tmpdir)
            self.assertEqual(len(rows), 1)
            self.assertEqual(rows[0]["name"], "job1")
    
    def test_collect_valid_status(self):
        with tempfile.TemporaryDirectory() as tmpdir:
            sub1 = os.path.join(tmpdir, "job1")
            os.makedirs(sub1)
            status = {"name": "job1", "state": "done", "reason": "ok", "coder_calls": 5, "updated": "2023-01-01"}
            with open(os.path.join(sub1, "status.json"), "w") as f:
                json.dump(status, f)
            rows, bad = loop_status.collect(tmpdir)
            self.assertEqual(len(rows), 1)
            self.assertEqual(rows[0]["name"], "job1")
            self.assertEqual(rows[0]["state"], "done")
            self.assertEqual(rows[0]["reason"], "ok")
            self.assertEqual(rows[0]["coder_calls"], 5)
            self.assertEqual(rows[0]["updated"], "2023-01-01")
            self.assertEqual(bad, [])
    
    def test_collect_missing_fields(self):
        with tempfile.TemporaryDirectory() as tmpdir:
            sub1 = os.path.join(tmpdir, "job1")
            os.makedirs(sub1)
            status = {}
            with open(os.path.join(sub1, "status.json"), "w") as f:
                json.dump(status, f)
            rows, bad = loop_status.collect(tmpdir)
            self.assertEqual(len(rows), 1)
            self.assertEqual(rows[0]["name"], "")
            self.assertEqual(rows[0]["state"], "")
            self.assertEqual(rows[0]["reason"], "")
            self.assertEqual(rows[0]["coder_calls"], 0)
            self.assertEqual(rows[0]["updated"], "")
    
    def test_collect_invalid_json(self):
        with tempfile.TemporaryDirectory() as tmpdir:
            sub1 = os.path.join(tmpdir, "job1")
            os.makedirs(sub1)
            with open(os.path.join(sub1, "status.json"), "w") as f:
                f.write("not json")
            rows, bad = loop_status.collect(tmpdir)
            self.assertEqual(rows, [])
            self.assertEqual(bad, ["job1"])
    
    def test_collect_not_dict(self):
        with tempfile.TemporaryDirectory() as tmpdir:
            sub1 = os.path.join(tmpdir, "job1")
            os.makedirs(sub1)
            with open(os.path.join(sub1, "status.json"), "w") as f:
                json.dump([1,2,3], f)
            rows, bad = loop_status.collect(tmpdir)
            self.assertEqual(rows, [])
            self.assertEqual(bad, ["job1"])
    
    def test_collect_unreadable_file(self):
        with tempfile.TemporaryDirectory() as tmpdir:
            sub1 = os.path.join(tmpdir, "job1")
            os.makedirs(sub1)
            os.makedirs(os.path.join(sub1, "status.json"))
            rows, bad = loop_status.collect(tmpdir)
            self.assertEqual(rows, [])
            self.assertEqual(bad, ["job1"])
    
    def test_collect_multiple_jobs_sorting(self):
        with tempfile.TemporaryDirectory() as tmpdir:
            for name, updated in [("job1", "2023-01-02"), ("job2", "2023-01-01"), ("job3", "2023-01-02")]:
                sub = os.path.join(tmpdir, name)
                os.makedirs(sub)
                status = {"name": name, "state": "done", "reason": "ok", "coder_calls": 0, "updated": updated}
                with open(os.path.join(sub, "status.json"), "w") as f:
                    json.dump(status, f)
            rows, bad = loop_status.collect(tmpdir)
            self.assertEqual(len(rows), 3)
            self.assertEqual(rows[0]["name"], "job1")
            self.assertEqual(rows[1]["name"], "job3")
            self.assertEqual(rows[2]["name"], "job2")
    
    def test_reason_truncation(self):
        with tempfile.TemporaryDirectory() as tmpdir:
            sub1 = os.path.join(tmpdir, "job1")
            os.makedirs(sub1)
            long_reason = "a" * 100
            status = {"name": "job1", "state": "done", "reason": long_reason, "coder_calls": 0, "updated": "2023-01-01"}
            with open(os.path.join(sub1, "status.json"), "w") as f:
                json.dump(status, f)
            rows, bad = loop_status.collect(tmpdir)
            self.assertEqual(len(rows), 1)
            self.assertEqual(len(rows[0]["reason"]), 80)
            self.assertEqual(rows[0]["reason"], "a" * 80)
    
    def test_bad_sorted(self):
        with tempfile.TemporaryDirectory() as tmpdir:
            for name in ["zebra", "apple", "mango"]:
                sub = os.path.join(tmpdir, name)
                os.makedirs(sub)
                with open(os.path.join(sub, "status.json"), "w") as f:
                    f.write("invalid")
            rows, bad = loop_status.collect(tmpdir)
            self.assertEqual(bad, ["apple", "mango", "zebra"])
    
    def test_main_output(self):
        with tempfile.TemporaryDirectory() as tmpdir:
            sub1 = os.path.join(tmpdir, "job1")
            os.makedirs(sub1)
            status = {"name": "job1", "state": "done", "reason": "ok", "coder_calls": 5, "updated": "2023-01-01"}
            with open(os.path.join(sub1, "status.json"), "w") as f:
                json.dump(status, f)
            old_stdout = sys.stdout
            sys.stdout = StringIO()
            try:
                ret = loop_status.main(["loop_status", tmpdir])
                output = sys.stdout.getvalue()
            finally:
                sys.stdout = old_stdout
            self.assertEqual(ret, 0)
            self.assertIn("job1 | done | 5 | ok", output)
    
    def test_main_output_with_bad(self):
        with tempfile.TemporaryDirectory() as tmpdir:
            sub1 = os.path.join(tmpdir, "good")
            os.makedirs(sub1)
            status = {"name": "good", "state": "done", "reason": "ok", "coder_calls": 1, "updated": "2023-01-01"}
            with open(os.path.join(sub1, "status.json"), "w") as f:
                json.dump(status, f)
            sub2 = os.path.join(tmpdir, "bad_job")
            os.makedirs(sub2)
            with open(os.path.join(sub2, "status.json"), "w") as f:
                f.write("invalid")
            old_stdout = sys.stdout
            sys.stdout = StringIO()
            try:
                ret = loop_status.main(["loop_status", tmpdir])
                output = sys.stdout.getvalue()
            finally:
                sys.stdout = old_stdout
            self.assertEqual(ret, 0)
            self.assertIn("good | done | 1 | ok", output)
            self.assertIn("unreadable: bad_job", output)
    
    def test_main_default_loop_dir(self):
        with tempfile.TemporaryDirectory() as tmpdir:
            loop_dir = os.path.join(tmpdir, "_loop")
            os.makedirs(loop_dir)
            sub1 = os.path.join(loop_dir, "job1")
            os.makedirs(sub1)
            status = {"name": "job1", "state": "done", "reason": "ok", "coder_calls": 5, "updated": "2023-01-01"}
            with open(os.path.join(sub1, "status.json"), "w") as f:
                json.dump(status, f)
            old_cwd = os.getcwd()
            os.chdir(tmpdir)
            try:
                old_stdout = sys.stdout
                sys.stdout = StringIO()
                try:
                    ret = loop_status.main(["loop_status"])
                    output = sys.stdout.getvalue()
                finally:
                    sys.stdout = old_stdout
                self.assertEqual(ret, 0)
                self.assertIn("job1 | done | 5 | ok", output)
            finally:
                os.chdir(old_cwd)

if __name__ == "__main__":
    unittest.main()
