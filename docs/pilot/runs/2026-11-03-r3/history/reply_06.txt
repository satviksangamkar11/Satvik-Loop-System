"""Loop status summary tool."""
import json
import os


def collect(loop_dir):
    rows = []
    bad = []
    if not os.path.isdir(loop_dir):
        return rows, bad
    for entry in os.listdir(loop_dir):
        full = os.path.join(loop_dir, entry)
        if not os.path.isdir(full):
            continue
        if not entry.endswith('status.json'):
            continue
        try:
            with open(full, 'r', encoding='utf-8') as fh:
                data = json.load(fh)
        except (OSError, ValueError):
            bad.append(entry)
            continue
        if not isinstance(data, dict):
            bad.append(entry)
            continue
        name = data.get('name')
        state = data.get('state')
        reason = data.get('reason') or ''
        updated = data.get('updated')
        if updated is None:
            updated = ''
        coder_calls = data.get('coder_calls')
        if coder_calls is None:
            coder_calls = 0
        else:
            coder_calls = int(coder_calls)
        rows.append({
            'name': name if name is not None else '',
            'state': state if state is not None else '',
            'reason': reason[:80],
            'coder_calls': coder_calls,
            'updated': updated,
        })
    rows.sort(key=lambda r: (r['updated'], r['name']))
    bad.sort()
    return rows, bad


def main(argv=None):
    if argv is None:
        argv = ['_loop']
    loop_dir = argv[0] if len(argv) > 0 else '_loop'
    rows, bad = collect(loop_dir)
    for row in rows:
        print('{} | {} | {} | {}'.format(row['name'], row['state'], row['coder_calls'], row['reason']))
    if bad:
        print('unreadable: ' + ', '.join(bad))
    return 0

import os
import sys
import json

import pytest

sys.path.insert(0, os.path.join(os.path.dirname(__file__), '..', 'scripts', 'source'))
import loop_status


def make_status(folder, **fields):
    path = os.path.join(folder, 'status.json')
    with open(path, 'w', encoding='utf-8') as fh:
        json.dump(fields, fh)
    return path


def test_missing_dir():
    assert loop_status.collect('/nonexistent/path/xyz') == ([], [])


def test_normal_order(tmp_path):
    f1 = tmp_path / 'a'
    f2 = tmp_path / 'b'
    make_status(str(f1), name='A', state='running', reason='ok', updated='2025-01-01', coder_calls=3)
    make_status(str(f2), name='B', state='done', reason='done', updated='2024-01-01', coder_calls=0)
    rows, bad = loop_status.collect(str(tmp_path))
    assert bad == []
    assert len(rows) == 2
    assert rows[0]['name'] == 'B'
    assert rows[0]['coder_calls'] == 0
    assert rows[0]['reason'] == 'done'
    assert rows[1]['name'] == 'A'


def test_invalid_json(tmp_path):
    bad_folder = tmp_path / 'bad'
    good_folder = tmp_path / 'good'
    bad_folder.mkdir()
    good_folder.mkdir()
    with open(bad_folder / 'status.json', 'w') as fh:
        fh.write('not json')
    make_status(str(good_folder), name='Y', updated='2025-01-01')
    rows, bad = loop_status.collect(str(tmp_path))
    assert bad == ['bad']
    assert len(rows) == 1
    assert rows[0]['name'] == 'Y'


def test_not_object(tmp_path):
    arr_folder = tmp_path / 'arr'
    arr_folder.mkdir()
    with open(arr_folder / 'status.json', 'w') as fh:
        fh.write('[1, 2, 3]')
    rows, bad = loop_status.collect(str(tmp_path))
    assert bad == ['arr']


def test_reason_cut(tmp_path):
    folder = tmp_path / 'long'
    folder.mkdir()
    make_status(str(folder), name='L', reason='x' * 100, updated='2025-01-01')
    rows, bad = loop_status.collect(str(tmp_path))
    assert len(rows) == 1
    assert rows[0]['reason'] == 'x' * 80


def test_missing_fields(tmp_path):
    folder = tmp_path / 'm'
    folder.mkdir()
    make_status(str(folder), name='M', state='S', updated='2025-01-01')
    rows, bad = loop_status.collect(str(tmp_path))
    assert len(rows) == 1
    assert rows[0]['reason'] == ''
    assert rows[0]['coder_calls'] == 0


def test_bad_sorted(tmp_path):
    for name in ['z', 'a', 'm']:
        folder = tmp_path / name
        folder.mkdir()
        with open(folder / 'status.json', 'w') as fh:
            fh.write('broken')
    rows, bad = loop_status.collect(str(tmp_path))
    assert bad == ['a', 'm', 'z']


def test_main_output(capsys, tmp_path):
    a = tmp_path / 'a'
    b = tmp_path / 'b'
    a.mkdir()
    b.mkdir()
    make_status(str(a), name='A', state='run', reason='r', updated='2025-01-01', coder_calls=2)
    make_status(str(b), name='B', state='done', reason='d', updated='2024-01-01', coder_calls=0)
    rc = loop_status.main([str(tmp_path)])
    captured = capsys.readouterr()
    lines = captured.out.strip().splitlines()
    assert rc == 0
    assert lines[0] == 'A | run | 2 | r'
    assert lines[1] == 'B | done | 0 | d'
    assert len(lines) == 2


def test_main_bad(capsys, tmp_path):
    bad = tmp_path / 'bad'
    bad.mkdir()
    with open(bad / 'status.json', 'w') as fh:
        fh.write('broken')
    rc = loop_status.main([str(tmp_path)])
    captured = capsys.readouterr()
    assert rc == 0
    assert captured.out.startswith('unreadable: bad')
