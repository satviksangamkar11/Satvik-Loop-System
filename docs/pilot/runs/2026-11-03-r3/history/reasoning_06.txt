```python
"""Loop status summary tool."""
import json
import os


def collect(loop_dir):
    rows = []
    bad = []
    if not os.path.isdir(loop_dir):
        return rows, bad
    for entry in os.listdir(loop_dir):
        full = os.path.join(loop_dir, entry)
        if not os.path.isfile(full):
            continue
        if not entry.endswith('status.json'):
            continue
        try:
            with open(full, 'r', encoding='utf-8') as fh:
                data = json.load(fh)
        except (OSError, ValueError):
            bad.append(entry)
            continue
        if not isinstance(data, dict):
            bad.append(entry)
            continue
        name = data.get('name')
        state = data.get('state')
        reason = data.get('reason')
        updated = data.get('updated')
        if updated is None:
            updated = ''
        coder_calls = data.get('coder_calls')
        if coder_calls is None:
            coder_calls = 0
        else:
            coder_calls = int(coder_calls)
        rows.append({
            'name': name if name is not None else '',
            'state': state if state is not None else '',
            'reason': reason[:80],
            'coder_calls': coder_calls,
            'updated': updated,
        })
    rows.sort(key=lambda r: (r['updated'], r['name']))
    bad.sort()
    return rows, bad


def main(argv=None):
    if argv is None:
        argv = ['_loop']
    loop_dir = argv[0] if len(argv) > 0 else '_loop'
    rows, bad = collect(loop_dir)
    for row in rows:
        print('{} | {} | {} | {}'.format(row['name'], row['state'], row['coder_calls'], row['reason']))
    if bad:
        print('unreadable: ' + ', '.join(bad))
    return 0
```
